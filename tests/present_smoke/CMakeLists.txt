cmake_minimum_required(VERSION 3.24)
project(PresentSmoke LANGUAGES CXX)
enable_testing()
add_executable(present_smoke present_smoke.cpp)
target_compile_features(present_smoke PRIVATE cxx_std_17)
target_compile_definitions(present_smoke PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(present_smoke PRIVATE ../../src)
target_link_libraries(present_smoke PRIVATE d3d11 d3d12 dxgi dxguid)
add_test(NAME warp_present COMMAND present_smoke)
set_tests_properties(warp_present PROPERTIES TIMEOUT 60 SKIP_RETURN_CODE 77)
